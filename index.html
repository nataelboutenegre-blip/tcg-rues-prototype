-- ===================================================================== -- INTENSITE DES ATTAQUES ET DES DEFENSES -- A executer apres tous les autres fichiers SQL. -- -- Avant chaque round, le joueur choisit son engagement : -- prudente : 35 %% du prix, 35 %% de chances -> le moins cher, mais tres lent -- normale : prix normal, 50 %% -> equilibre -- offensive : prix double, 65 %% -> deux fois plus rapide, un peu plus cher -- ===================================================================== begin; create or replace function chance_intensite(p_intensite text) returns numeric language sql immutable as $$ select case p_intensite when 'prudente' then 0.35 when 'normale' then 0.50 when 'offensive' then 0.65 end; $$; create or replace function cout_intensite(p_intensite text) returns numeric language sql immutable as $$ select case p_intensite when 'prudente' then 0.35 when 'normale' then 1.0 when 'offensive' then 2.0 end; $$; -- les anciennes versions (sans intensite) sont remplacees par celles-ci drop function if exists attaquer(text); drop function if exists defendre(text, uuid); create or replace function attaquer(p_commune_code text, p_intensite text default 'normale') returns table(gagne boolean, victoires_consecutives integer, conquise boolean, cout integer, chances integer) language plpgsql security definer as $$ #variable_conflict use_column declare v_attaquant_id uuid := auth.uid(); v_tier text; v_defenseur_id uuid; v_cout integer; v_solde integer; v_derniere_acquisition timestamptz; v_bouclier_jusqua timestamptz; v_siege_actuel integer; v_dernier_round timestamptz; v_gagne boolean; v_nouvelles_victoires integer; v_bonus integer; v_chance numeric; begin if v_attaquant_id is null then raise exception 'Non connecte'; end if; select tier into v_tier from communes where code = p_commune_code; if v_tier is null then raise exception 'Commune inconnue'; end if; if v_tier not in ('rare', 'legendaire') then raise exception 'Seules les communes rares et legendaires peuvent etre attaquees'; end if; select p.joueur_id, p.acquired_at, p.bouclier_jusqua into v_defenseur_id, v_derniere_acquisition, v_bouclier_jusqua from possessions p where p.commune_code = p_commune_code; if v_defenseur_id is null then raise exception 'Cette commune n''appartient a personne'; end if; if v_defenseur_id = v_attaquant_id then raise exception 'Tu ne peux pas attaquer ta propre commune'; end if; if v_derniere_acquisition > now() - interval '3 hours' then raise exception 'Cette commune est encore protegee (immunite 3h apres acquisition)'; end if; if v_bouclier_jusqua is not null and v_bouclier_jusqua > now() then raise exception 'Cette commune est protégée par un bouclier. Ta série est conservée et reprendra à la fin.'; end if; select s.victoires_consecutives, s.dernier_round into v_siege_actuel, v_dernier_round from sieges s where s.attacker_id = v_attaquant_id and s.commune_code = p_commune_code; if v_dernier_round is not null and v_dernier_round > now() - delai_attaque(v_tier) then raise exception 'Tu dois attendre encore un peu avant de retenter sur cette cible'; end if; if v_siege_actuel is null then v_siege_actuel := 0; end if; -- intensite choisie par l'attaquant : plus on paie, plus on a de chances v_chance := chance_intensite(p_intensite); if v_chance is null then raise exception 'Intensité inconnue'; end if; v_cout := greatest(1, round(cout_attaque(v_tier) * cout_intensite(p_intensite)))::integer; select solde into v_solde from joueurs where id = v_attaquant_id; if v_solde < v_cout then raise exception 'Solde insuffisant pour attaquer (% points requis)', v_cout; end if; update joueurs set solde = solde - v_cout where id = v_attaquant_id; v_gagne := random() < v_chance; if v_gagne then v_nouvelles_victoires := v_siege_actuel + 1; else v_nouvelles_victoires := 0; end if; insert into sieges (attacker_id, commune_code, victoires_consecutives, dernier_round) values (v_attaquant_id, p_commune_code, v_nouvelles_victoires, now()) on conflict (attacker_id, commune_code) do update set victoires_consecutives = v_nouvelles_victoires, dernier_round = now(), -- une defaite termine l'attaque : le defenseur retrouve sa defense pour la prochaine defense_utilisee = case when v_nouvelles_victoires = 0 then false else sieges.defense_utilisee end; if v_nouvelles_victoires >= 3 then -- bonus de conquete : la moitie de la valeur de rachat v_bonus := prix_rachat(v_tier) / 2; update possessions set joueur_id = v_attaquant_id, acquired_at = now(), conquise_le = now() where commune_code = p_commune_code; update joueurs set solde = solde + v_bonus where id = v_attaquant_id; update historique set released_at = now() where commune_code = p_commune_code and joueur_id = v_defenseur_id and released_at is null; insert into historique (commune_code, joueur_id, saison, acquired_at) values (p_commune_code, v_attaquant_id, 'saison-1', now()); delete from sieges where attacker_id = v_attaquant_id and commune_code = p_commune_code; return query select true, 3, true, v_cout, round(v_chance * 100)::integer; else return query select v_gagne, v_nouvelles_victoires, false, v_cout, round(v_chance * 100)::integer; end if; end; $$; -- Defense : le proprietaire tente de retirer une victoire a un attaquant create or replace function defendre(p_commune_code text, p_attaquant uuid, p_intensite text default 'normale') returns table(reussie boolean, victoires_restantes integer, cout integer, chances integer) language plpgsql security definer set search_path = public as $$ #variable_conflict use_column declare v_uid uuid := auth.uid(); v_tier text; v_proprio uuid; v_bouclier timestamptz; v_victoires integer; v_deja boolean; v_cout integer; v_solde integer; v_reussie boolean; v_restantes integer; v_chance numeric; begin if v_uid is null then raise exception 'Non connecté'; end if; select c.tier, p.joueur_id, p.bouclier_jusqua into v_tier, v_proprio, v_bouclier from possessions p join communes c on c.code = p.commune_code where p.commune_code = p_commune_code for update of p; if v_proprio is null or v_proprio <> v_uid then raise exception 'Cette commune ne t''appartient pas'; end if; if v_bouclier is not null and v_bouclier > now() then raise exception 'Ta commune est déjà protégée par un bouclier'; end if; select s.victoires_consecutives, s.defense_utilisee into v_victoires, v_deja from sieges s where s.commune_code = p_commune_code and s.attacker_id = p_attaquant for update of s; if v_victoires is null or v_victoires = 0 then raise exception 'Cet attaquant n''a aucune victoire en cours sur ta commune'; end if; if v_deja then raise exception 'Tu as déjà utilisé ta défense contre cette attaque'; end if; v_chance := chance_intensite(p_intensite); if v_chance is null then raise exception 'Intensité inconnue'; end if; v_cout := greatest(1, round(cout_attaque(v_tier) * cout_intensite(p_intensite)))::integer; select solde into v_solde from joueurs where id = v_uid for update; if v_solde < v_cout then raise exception 'Solde insuffisant pour défendre (% points requis)', v_cout; end if; update joueurs set solde = solde - v_cout where id = v_uid; v_reussie := random() < v_chance; if v_reussie then v_restantes := v_victoires - 1; else v_restantes := v_victoires; end if; -- si l'attaquant retombe a zero, son attaque est terminee : la defense sera de nouveau disponible update sieges set victoires_consecutives = v_restantes, defense_utilisee = (v_restantes > 0) where commune_code = p_commune_code and attacker_id = p_attaquant; return query select v_reussie, v_restantes, v_cout, round(v_chance * 100)::integer; end; $$; revoke all on function attaquer(text, text), defendre(text, uuid, text), chance_intensite(text), cout_intensite(text) from public, anon, authenticated; grant execute on function attaquer(text, text), defendre(text, uuid, text) to authenticated; commit;
